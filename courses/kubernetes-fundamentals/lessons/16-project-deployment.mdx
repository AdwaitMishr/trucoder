      ## The canonical deployment

      This is the manifest you would write for the API tier. Read every
      line — the tests will make you find the details:

      ```yaml
      apiVersion: apps/v1
      kind: Deployment
      metadata:
        name: shop-api
      spec:
        replicas: 3
        selector:
          matchLabels:
            app: shop-api
        template:
          metadata:
            labels:
              app: shop-api
          spec:
            containers:
              - name: shop-api
                image: ghcr.io/adith2005-20/shop-api:v1
                ports:
                  - containerPort: 8080
                readinessProbe:
                  httpGet:
                    path: /readyz
                    port: 8080
                  initialDelaySeconds: 5
                  periodSeconds: 10
                livenessProbe:
                  httpGet:
                    path: /healthz
                    port: 8080
                envFrom:
                  - configMapRef:
                      name: shop-config
                  - secretRef:
                      name: shop-secrets
                resources:
                  requests:
                    cpu: 100m
                    memory: 128Mi
                  limits:
                    cpu: 500m
                    memory: 512Mi
                volumeMounts:
                  - name: reports
                    mountPath: /data
            volumes:
              - name: reports
                persistentVolumeClaim:
                  claimName: shop-reports
      ```

      :::tip
      The selector `matchLabels: {app: shop-api}` must be an exact subset
      of the template's `labels` — the Deployment controller uses it to
      claim pods, the Service uses the same labels to route. Change one
      side and the Deployment creates pods nobody can find.
      :::

      :::warning
      A missing `readinessProbe` means the Service sends traffic to a pod
      that is still starting. A missing `livenessProbe` means a hung pod
      is never restarted. Both are production requirements, not
      decorations.
      :::

    hints:
      - "yaml.safe_load(manifest) gives you a dict. Containers live at spec.template.spec.containers."
      - "Compare the selector dict against the pod labels dict: every selector key must exist in the labels with the same value."
      - "A missing probe means the key is absent — use .get() so you can return None."
    solution: |
      import yaml

      def solve(manifest: str) -> list:
          d = yaml.safe_load(manifest)
          spec = d["spec"]
          selector = spec["selector"]["matchLabels"]
          labels = spec["template"]["metadata"]["labels"]
          selector_ok = all(labels.get(k) == v for k, v in selector.items())
          c = spec["template"]["spec"]["containers"][0]
          image = c["image"]
          replicas = spec.get("replicas", 1)
          readiness = c.get("readinessProbe", {}).get("httpGet", {}).get("path")
          liveness = c.get("livenessProbe", {}).get("httpGet", {}).get("path")
          return [image, replicas, selector_ok, readiness, liveness]
